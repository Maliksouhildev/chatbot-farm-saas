  // High-contrast, vibrant animated typing indicator (writing line animation)
  const renderTypingIndicator = (accentClass = 'text-emerald-600 dark:text-emerald-400') => {
    if (!isTyping) return null;
    return (
      <div className="flex items-center gap-2 py-1 px-3 rounded-full bg-black/5 dark:bg-white/10 w-fit select-none animate-in fade-in duration-200 my-1">
        <span className="flex items-center gap-1">
          <span className="w-1.5 h-1.5 rounded-full bg-emerald-500 dark:bg-emerald-400 animate-bounce [animation-delay:0ms]" />
          <span className="w-1.5 h-1.5 rounded-full bg-emerald-500 dark:bg-emerald-400 animate-bounce [animation-delay:150ms]" />
          <span className="w-1.5 h-1.5 rounded-full bg-emerald-500 dark:bg-emerald-400 animate-bounce [animation-delay:300ms]" />
        </span>
        <span className={`text-[11px] font-semibold ${accentClass}`}>
          {activeContact.name} is typing...
        </span>
      </div>
    );
  };

  // Full-featured Categorized Emoji Picker Panel
  const renderEmojiPickerPanel = () => (
    <AnimatePresence>
      {isEmojiPickerOpen && (
        <motion.div
          initial={{ opacity: 0, height: 0 }}
          animate={{ opacity: 1, height: 'auto' }}
          exit={{ opacity: 0, height: 0 }}
          transition={{ duration: 0.15 }}
          className="bg-white/95 dark:bg-[#1E222A]/95 backdrop-blur-md border-t border-[#DFDFD4] dark:border-[#2E333D] px-3 py-2 shrink-0 overflow-hidden shadow-lg select-none z-30"
        >
          <div className="flex items-center justify-between mb-2 pb-1.5 border-b border-gray-100 dark:border-neutral-800">
            <div className="flex items-center gap-1 overflow-x-auto no-scrollbar">
              {EMOJI_CATEGORIES.map((cat) => (
                <button
                  key={cat.id}
                  type="button"
                  onClick={() => setActiveEmojiCategory(cat.id)}
                  className={`px-2.5 py-1 rounded-full text-xs font-semibold flex items-center gap-1 transition-all cursor-pointer ${
                    activeEmojiCategory === cat.id
                      ? 'bg-emerald-500 text-white shadow-2xs font-bold scale-102'
                      : 'bg-black/5 dark:bg-white/10 text-gray-700 dark:text-gray-300 hover:bg-black/10 dark:hover:bg-white/20'
                  }`}
                >
                  <span className="text-xs">{cat.icon}</span>
                  <span className="text-[11px]">{cat.name}</span>
                </button>
              ))}
            </div>
            <button
              type="button"
              onClick={() => setIsEmojiPickerOpen(false)}
              className="p-1 hover:bg-gray-100 dark:hover:bg-neutral-800 rounded-full text-gray-400 hover:text-gray-600 dark:hover:text-gray-200 cursor-pointer ml-2"
              title="Close emojis"
            >
              <X className="w-3.5 h-3.5" />
            </button>
          </div>
          <div className="grid grid-cols-8 sm:grid-cols-12 gap-1.5 max-h-36 overflow-y-auto custom-scrollbar p-1">
            {(EMOJI_CATEGORIES.find((c) => c.id === activeEmojiCategory) || EMOJI_CATEGORIES[0]).emojis.map((emoji, idx) => (
              <button
                key={`${emoji}-${idx}`}
                type="button"
                onClick={() => setInputMessage((prev) => prev + emoji)}
                className="w-8 h-8 rounded-lg hover:bg-black/5 dark:hover:bg-white/10 flex items-center justify-center text-lg hover:scale-125 transition-transform active:scale-95 cursor-pointer shadow-2xs"
              >
                {emoji}
              </button>
            ))}
          </div>
        </motion.div>
      )}
    </AnimatePresence>
  );

  // Dynamic Quick Responses Bar with Pinned / Sticky (+) Custom Template Creator synced across ALL channels
  const renderQuickTemplatesBar = () => {
    if (isChatReadOnly) return null;
    return (
      <>
        <div className="relative px-3 py-1.5 bg-[#F0F2F5] dark:bg-[#111B21] border-t border-[#DFDFD4] dark:border-[#2E333D] flex items-center gap-2 shrink-0 select-none">
          {/* Left Category Label */}
          <div className="flex items-center gap-1.5 shrink-0 pr-1.5 border-r border-gray-300/60 dark:border-neutral-700/60">
            <Zap className="w-3.5 h-3.5 text-[#1B6648] dark:text-emerald-400 shrink-0" />
            <span className="text-[10px] font-extrabold uppercase text-gray-600 dark:text-gray-300 shrink-0 tracking-wider hidden sm:inline">
              RÉPONSES RAPIDES
            </span>
          </div>

          {/* Scrollable Templates Stream */}
          <div 
            id="quick-templates-scroll-stream"
            onWheel={(e) => {
              if (e.deltaY !== 0) {
                e.currentTarget.scrollLeft += e.deltaY;
              }
            }}
            className="flex-1 min-w-0 flex items-center gap-1.5 overflow-x-auto no-scrollbar py-0.5 scroll-smooth touch-pan-x"
          >
            {quickTemplates.map((t) => {
              const resolvedText = t.text.replace(/\{name\}/g, activeContact?.name || '');
              return (
                <div key={t.id} className="relative group/tpl shrink-0 flex items-center">
                  <button
                    type="button"
                    onClick={() => setInputMessage(resolvedText)}
                    className="text-xs px-2.5 py-1 rounded-full bg-white dark:bg-[#1E293B] text-gray-800 dark:text-gray-100 shadow-2xs border border-gray-200 dark:border-slate-700/80 hover:border-[#1B6648] dark:hover:border-emerald-400 hover:bg-gray-50 dark:hover:bg-[#27354A] font-medium transition-all active:scale-95 cursor-pointer flex items-center gap-1.5 whitespace-nowrap"
                    title={resolvedText}
                  >
                    <span className="text-sm leading-none">{t.icon || '💬'}</span>
                    <span className="text-gray-900 dark:text-gray-100 font-semibold">{t.label}</span>
                  </button>
                  {t.isCustom && (
                    <button
                      type="button"
                      onClick={(e) => handleDeleteCustomTemplate(t.id, e)}
                      className="opacity-0 group-hover/tpl:opacity-100 p-0.5 -ml-2 mr-1 bg-red-500 hover:bg-red-600 text-white rounded-full transition-opacity cursor-pointer shadow-2xs"
                      title="Supprimer ce modèle personnalisé"
                    >
                      <X className="w-2.5 h-2.5" />
                    </button>
                  )}
                </div>
              );
            })}
          </div>

          {/* Pinned / Sticky Add Premade Message Button (Always docked & visible regardless of message count or scroll) */}
          <div className="sticky right-0 shrink-0 flex items-center pl-2 border-l border-gray-300/60 dark:border-neutral-700/60 bg-[#F0F2F5] dark:bg-[#111B21] shadow-[-8px_0_10px_-3px_rgba(0,0,0,0.06)] dark:shadow-[-8px_0_10px_-3px_rgba(0,0,0,0.4)] z-10">
            <button
              id="sticky-add-template-btn"
              type="button"
              onClick={() => setIsTemplateModalOpen(true)}
              className="text-xs px-3 py-1 rounded-full bg-[#1B6648] hover:bg-[#155239] text-white dark:bg-emerald-600 dark:hover:bg-emerald-500 font-bold transition-all active:scale-95 cursor-pointer flex items-center gap-1.5 shrink-0 whitespace-nowrap shadow-xs"
              title="Ajouter un message préenregistré personnalisé"
            >
              <Plus className="w-3.5 h-3.5" />
              <span>Ajouter</span>
            </button>
          </div>
        </div>

        {/* Custom Template Creator Modal */}
        <AnimatePresence>
          {isTemplateModalOpen && (
            <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/50 backdrop-blur-xs animate-in fade-in duration-150">
              <motion.div
                initial={{ scale: 0.95, opacity: 0 }}
                animate={{ scale: 1, opacity: 1 }}
                exit={{ scale: 0.95, opacity: 0 }}
                className="bg-white dark:bg-[#1E222A] rounded-2xl shadow-2xl border border-gray-200 dark:border-neutral-700 p-5 w-full max-w-md space-y-4"
              >
                <div className="flex items-center justify-between border-b border-gray-100 dark:border-neutral-800 pb-3">
                  <div className="flex items-center gap-2">
                    <span className="p-2 rounded-xl bg-emerald-50 dark:bg-emerald-950/40 text-[#1B6648] dark:text-emerald-400">
                      <Plus className="w-4 h-4" />
                    </span>
                    <div>
                      <h3 className="font-bold text-sm text-gray-900 dark:text-gray-100">Nouveau message préenregistré</h3>
                      <p className="text-[11px] text-gray-500 dark:text-gray-400">Créez une réponse rapide personnalisée</p>
                    </div>
                  </div>
                  <button
                    type="button"
                    onClick={() => setIsTemplateModalOpen(false)}
                    className="p-1 rounded-full hover:bg-gray-100 dark:hover:bg-neutral-800 text-gray-400 hover:text-gray-600 dark:hover:text-gray-200 transition-colors cursor-pointer"
                  >
                    <X className="w-4 h-4" />
                  </button>
                </div>

                <form onSubmit={handleSaveCustomTemplate} className="space-y-3.5">
                  <div className="grid grid-cols-4 gap-2">
                    <div className="col-span-1 space-y-1">
                      <label className="text-[11px] font-semibold text-gray-700 dark:text-gray-200">Icône</label>
                      <input
                        type="text"
                        maxLength={4}
                        value={newTemplateIcon}
                        onChange={(e) => setNewTemplateIcon(e.target.value)}
                        placeholder="💬"
                        className="w-full text-center px-2 py-1.5 rounded-xl border border-gray-200 dark:border-neutral-700 bg-gray-50 dark:bg-neutral-800 text-sm text-gray-900 dark:text-white placeholder-gray-400 caret-emerald-500 focus:outline-hidden focus:ring-2 focus:ring-[#1B6648] dark:focus:ring-emerald-500"
                      />
                    </div>
                    <div className="col-span-3 space-y-1">
                      <label className="text-[11px] font-semibold text-gray-700 dark:text-gray-200">Titre du bouton</label>
                      <input
                        type="text"
                        required
                        value={newTemplateLabel}
                        onChange={(e) => setNewTemplateLabel(e.target.value)}
                        placeholder="Ex: Devis Rapide"
                        className="w-full px-3 py-1.5 rounded-xl border border-gray-200 dark:border-neutral-700 bg-gray-50 dark:bg-neutral-800 text-sm text-gray-900 dark:text-white placeholder-gray-400 caret-emerald-500 focus:outline-hidden focus:ring-2 focus:ring-[#1B6648] dark:focus:ring-emerald-500"
                      />
                    </div>
                  </div>

                  <div className="space-y-1">
                    <div className="flex items-center justify-between">
                      <label className="text-[11px] font-semibold text-gray-700 dark:text-gray-200">Message</label>
                      <span className="text-[10px] text-emerald-600 dark:text-emerald-400 font-mono">
                        Conseil: tapez {'{name}'} pour le nom client
                      </span>
                    </div>
                    <textarea
                      required
                      rows={3}
                      value={newTemplateText}
                      onChange={(e) => setNewTemplateText(e.target.value)}
                      placeholder="Ex: Bonjour {name}, voici les détails de votre demande..."
                      className="w-full px-3 py-2 rounded-xl border border-gray-200 dark:border-neutral-700 bg-gray-50 dark:bg-neutral-800 text-xs text-gray-900 dark:text-white placeholder-gray-400 caret-emerald-500 focus:outline-hidden focus:ring-2 focus:ring-[#1B6648] dark:focus:ring-emerald-500 resize-none"
                    />
                  </div>

                  <div className="flex items-center justify-end gap-2 pt-2">
                    <button
                      type="button"
                      onClick={() => setIsTemplateModalOpen(false)}
                      className="px-3 py-1.5 rounded-xl text-xs text-gray-600 dark:text-gray-300 hover:bg-gray-100 dark:hover:bg-neutral-800 font-medium transition-colors cursor-pointer"
                    >
                      Annuler
                    </button>
                    <button
                      type="submit"
                      className="px-4 py-1.5 rounded-xl text-xs bg-[#1B6648] hover:bg-[#144f37] text-white font-semibold shadow-xs transition-colors cursor-pointer"
                    >
                      Enregistrer
                    </button>
                  </div>
                </form>
              </motion.div>
            </div>
          )}
        </AnimatePresence>
      </>
    );
  };
